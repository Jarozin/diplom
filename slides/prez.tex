\begin{frame}
    \frametitle{Формализация задачи}
    Граф инструкции $ G = (V, A, E) $, где:
    \begin{itemize}
        \item[---] $ V = \{s_1, s_2, \dots, s_n\} $ --- множество состояний;
        \item[---] $A$ ---  множество действий;
        \item[---] $ E \subseteq V \times A \times V $ --- множество переходов (действий).
    \end{itemize}
    \vspace{0.2cm}
    Каждое состояние $ S_i $ характеризуется:
    \begin{itemize}
        \item[---] $ name_i $ --- название состояния;
        \item[---] $ desc_i $ --- описание;
        \item[---] $ type_i $ --- тип (начальное, промежуточное, конечное, ошибка);
        \item[---] $ objects\_state_i $ --- состояние объектов в данном состоянии.
    \end{itemize}
    \vspace{0.2cm}
    Требуется построить отображение:
    \[
        F: S \times UserAction \to Recommendation,
    \]
    где каждому действию ставится в соответствие набор рекомендаций, ранжированных по важности.
\end{frame}

\begin{frame}
    \frametitle{Метод анализа иерархий (МАИ)}
    \textbf{Классический МАИ.}
    \begin{itemize}
        \item[---] Построение иерархии: цель, критерии, альтернативы.
        \item[---] Парные сравнения элементов по шкале Саати (1---9).
        \item[---] Вычисление весов через геометрическое среднее.
        \item[---] Проверка согласованности: $CR = CI/RI < 0.1$.
    \end{itemize}
    \vfill
    \textbf{Модифицированный МАИ.}
    \begin{itemize}
        \item[---] Группировка альтернатив.
        \item[---] Сравнения только внутри групп.
        \item[---] Рекурсивный жадный отбор лучших $N$ альтернатив.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Алгоритм ранжирования рекомендаций}
    \textbf{Вход:} действие \(a\) (его название и описание), набор меток \(L\) с рекомендациями \(R_a = \{r_1,\dots,r_m\}\), веса критериев \(w_c\), количество искомых лучших рекомендаций $n$.
    \vspace{0.2cm}

    \textbf{Алгоритм:}
    \begin{enumerate}
        \item[---] Инициализировать пустой список оценок \(S\).
        \item[---] Для каждой рекомендации \(r_i\):
        \begin{itemize}
                \item[---] получить оценки по критериям \(s_{i,c}\) из конфигурации (если отсутствуют, использовать значение по умолчанию 0.5).
                \item[---] вычислить интегральную оценку \(S_i = \sum_{c} w_c \cdot s_{i,c}\).
            \end{itemize}
            \item[---] Вычислить сумму \(T = \sum_{i=1}^{m} S_i\).
            \item[---] Если \(T > 0\), нормализовать \(\tilde{S}_i = S_i / T\); иначе \(\tilde{S}_i = 1/m\).
        \item[---] Отсортировать рекомендации по убыванию \(\tilde{S}_i\).
        \item[---] Вернуть n лучших элементов с их оценками.
    \end{enumerate}
    \vspace{0.2cm}
    \textbf{Выход:} список рекомендаций, отсортированный по убыванию нормализованной оценки.
\end{frame}
