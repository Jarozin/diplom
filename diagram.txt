    BackgroundColor #E0E0E0

    BackgroundColor #E0E0E0
    BorderColor #757575
    FontColor #212121
}

    ArrowColor #424242
    LifeLineBorderColor #9E9E9E
}

actor "Оператор" as u
participant "Компонент сопровождения оператора" as f
participant "Компонент работы с графами" as g
participant "Компонент обработки аварийных ситуаций" as a
participant "Компонент работы с контекстом" as c
participant "Компонент ранжирования рекомендаций" as r
u -> f : Данные об операторе
f -> f : Обработать данные об уровне экспертизы оператора
f -> r : Уровень экспертизы оператора
r -> r : Установка весов критериев \nв соответствии с \nуровнем экспертизы
f -->> u : Запрос выбора инструкции
u -> f : Выбор инструкции в графовом формате
f -> g : Запрос данных графовой инструкции
g -> g : Выгрузить и предобработать данные инструкции
g -->> f : Данные инструкции
f -> f : Определить начальное состояние и возможное действие
loop
f -> c : Запросить подходящие рекомендации для действия в текущем состоянии
c -> c : Определить подходящие \nконтекстуальные метки и \nсоответствующие рекомендации
c -> r : Запрос ранжирования набора рекомендаций
r -> r : Ранжирование \nрекомендаций
r -->> c : Ранжированный список рекомендаций
c -->> f : Контекстуальные метки и кортеж наиболее релевантных рекомендаций
f -->> u : Вывод рекомендаций к действию
f -> u : Запрос результата действия
u -->> f : Результат совершённого действия
opt Действие приводит к аварии
  f -> f : Сохранить текущее состояние перед сменой графа
  f -> a : Запрос обработки аварийного сценария с учётом контекстуальных меток
  a -> a : Определить подходящие графы обработки аварийных ситуаций
  a -> g : Запрос загрузки подходящих графов аварийных сценариев
  g -> g : Загрузка  графов аварийных сценариев
  g -->> a : Графы аварийных сценариев
  a -> a : Определить начальное состояние и возможное действие
loop
a -> c : Запросить подходящие рекомендации для действия в текущем состоянии
c -> c : Определить подходящие \nконтекстуальные метки и \nсоответствующие рекомендации
c -> r : Запрос ранжирования набора рекомендаций
r -> r : Ранжирование \nрекомендаций
r -->> c : Ранжированный список рекомендаций
c -->> a : Контекстуальные метки и кортеж наиболее релевантных рекомендаций
a -->> u : Вывод рекомендаций к действию
a -> u : Запрос результата действия
u -->> a : Результат совершённого действия
alt Текущее состояние конечное
  alt Аварийная ситуация позволяет возврат
    a -->> f: Возврат к выполнению графа инструкции
  else Аварийная ситуация невозвратима
    a -->> u: Достигнуто конечное состояние
  end
else Текущее состояние промежуточное
  a -> a : Переход к следующему шагу  
end
end
f -> f : Определение текущего состояния и возможных действий
alt Текущее состояние конечное
  f -->> u : Достигнуто конечное состояние
else Текущее состояние промежуточное
  f -> f : Переход к следующему шагу  
end
end
