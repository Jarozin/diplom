@startuml
skinparam defaultFontSize 19

skinparam backgroundColor #FFFFFF
skinparam actor {
  BackgroundColor #FFFFFF
  BorderColor #000000
  FontColor #212121
}
skinparam participant {
  BackgroundColor #E0E0E0
  BorderColor #E0E0E0
  FontColor #212121
}
skinparam sequence {
  ArrowColor #424242
  LifeLineBorderColor #9E9E9E
}

actor "Оператор" as u
participant "Компонент сопровождения оператора    " as f
participant "Компонент работы с графами   " as g
participant "Компонент обработки аварийных ситуаций    " as a
participant "Компонент работы с контекстом    " as c
participant "Компонент ранжирования рекомендаций   " as r

u -> f : Данные об операторе
activate f
f -> f : Обработать данные об уровне \nэкспертизы оператора
f -> r : Уровень экспертизы оператора
activate r
r -> r : Установка \nвесов критериев \nв соответствии с \nуровнем экспертизы
deactivate r

f -->> u : Запрос выбора инструкции
u -> f : Выбор инструкции в графовом формате\t
f -> g : Запрос данных графовой инструкции\t\t\t
activate g
g -> g : Загрузить и предобработать \nданные инструкции
g -->> f : Данные инструкции
deactivate g
f -> f : Определить начальное состояние \nи возможное действие
loop
  f -> c : Запросить подходящие рекомендации \nдля действия в текущем состоянии
activate c
  c -> c : Определить подходящие контекстуальные \nметки и соответствующие рекомендации
  c -> r : Запрос ранжирования набора рекомендаций
activate r
  r -> r : Ранжирование \nрекомендаций
  r -->> c : Ранжированный список рекомендаций
deactivate r
  c -->> f : Контекстуальные метки и кортеж \nнаиболее релевантных рекомендаций
deactivate c
  f -> u : Вывод рекомендаций к действию
  f -> u : Запрос результата действия
  u -->> f : Результат совершённого действия
  opt Действие приводит к аварии
    f -> f : Сохранить текущее состояние \nперед сменой графа
    f -> a : Запрос обработки аварийного сценария \nс учётом контекстуальных меток
activate a
    a -> a : Определить подходящие графы обработки аварийных ситуаций
    a -> g : Запрос загрузки подходящих \nграфов аварийных сценариев 
activate g  
    g -> g : Загрузка графов \nаварийных сценариев
    g -->> a : Графы аварийных сценариев
deactivate g
    a -> a : Определить начальное состояние и возможное действие
    loop
      a -> c : Запросить подходящие рекомендации для действия в текущем состоянии\t\t
activate c
      c -> c : Определить подходящие контекстуальные \nметки и соответствующие рекомендации\t\t
      c -> r : Запрос ранжирования набора рекомендаций
activate r
      r -> r : Ранжирование \nрекомендаций
      r -->> c : Ранжированный список рекомендаций
deactivate r
      c -->> a : Контекстуальные метки и кортеж наиболее релевантных рекомендаций
deactivate c
      a -> u : Вывод рекомендаций к действию
      a -> u : Запрос результата действия
      u -->> a : Результат совершённого действия
      a -> a : Определение текущего состояния и возможных действий
      alt Текущее состояние промежуточное
       a -> a : Переход к следующему шагу 
      else Текущее состояние аварийного сценария конечное
       alt Аварийная ситуация допускает возврат к выполнению инструкции 
        a -->> f: Возврат к выполнению графа инструкции
       else Аварийная ситуация необратима
        a -->> u: Завершение выполнения
      end
      deactivate a
    end
	f -> f : Определение текущего состояния \nи возможных действий\t

	alt Текущее состояние промежуточное
	 f -> f : Переход к следующему шагу 
	else Текущее состояние конечное
	 f -->> u : Достигнуто конечное состояние
deactivate f
    end
  end
end


@enduml
